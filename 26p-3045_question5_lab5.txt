#include<stdio.h>
int main()
{
int choice;
int quantity;
int total_items = 0;
float bill = 0;

do{
printf("Choice Item price (Rs)\n");
printf("1 Tea 60\n");
printf("2 Samosa 40\n");
printf("3 Sandwich 180\n");
printf("4 Checkout\n");

printf("Enter choice: ");
scanf("%d" , &choice);

if(choice>=1 && choice<=3) 
{
printf("Quantity: ");
scanf("%d", &quantity);
total_items=total_items+quantity;

if(choice == 1)
{
bill=bill+quantity * 40;
}
else
if(choice == 3)
{
bill=bill+quantity * 180;
}
printf("current bill: Rs %.2f\n", bill);
}
else
if(choice == 4)
{
break;
}
else
{
printf("Invalid choice. Try again\n");
}}
while(1);
    float discount = 0;
if(bill > 1000)
{
discount = bill*0.10;
}
float payable = bill - discount;
printf("Total Items: %d\n", total_items);
printf("Bill: RS %f\n", bill);
printf("Discount: Rs %f\n", discount);
printf("payyable: Rs %f\n", payable);

return 0;
}
