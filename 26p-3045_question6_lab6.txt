#include<stdio.h>

int main()

{
int rows,i,j;

printf("Enter the number of rows: ");
scanf("%d", &rows);

for(i=0;j<=rows;i++)
{
for(j=1;j<=rows-i;j++)
{
printf(" ");
}
for(j=0;j<=i;j++)
{
printf("%d ", i);
}
printf("\n");
}
for(i=rows-1;i>=0;i--)
{
for(j=1;j<=rows-i;j++)
{
printf(" ");
}
for(j=0;j<=i;j++)
{
printf("%d ", i);
}
printf("\n");
}
return 0;
}
