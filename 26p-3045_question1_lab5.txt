#include<stdio.h>
int main()
{
int runs;
int total_runs = 0;
int boundaries = 0;
int dot_balls = 0;

for(int i=1; i<=6; i++)
{
printf("Ball %d runs: ", i);
scanf("%d", &runs);

total_runs=total_runs+runs;


if(runs == 4||runs == 6)
{ 
boundaries++;
}
if(runs == 0)
{
dot_balls++;
} }
printf("Total runs: %d\n", total_runs);
printf("Boundaries: %d\n", boundaries);
printf("Dot balls: %d\n", dot_balls);
return 0;
}
