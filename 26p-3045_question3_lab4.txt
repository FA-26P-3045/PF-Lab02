#include<stdio.h>
int main()
{
int x,y;
printf("Enter the x\n");
scanf("%d",&x);
printf("Enter the y\n");
scanf("%d",&y);

if(x>0 && y>0)
printf("1st Quardrant");

else if(x<0 && y>0)
printf("2nd Quardrant");

else if(x<0 && y<0)
printf("3rd Quardrant");

else if(x>0 && y<0)
printf("4th Quardrant");
    
else if(x==0 && y==0)
printf("Origin");
    
else if(y==0)
printf("on x-axis");
    
else if(x==0)
printf("on y-axis");
return 0;
}
