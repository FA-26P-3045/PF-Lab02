#include<stdio.h>
int main()
{
int i,j,sum;
int count = 0;
int n = 1;

for (n = 1; count < 4; n++) 
{
sum = 0;
for (i = 1; i < n; i++) 
{
if (n % i == 0) 
{
sum = sum + i;
}}
if (sum == n) 
{
printf("%d is a perfect number\n", n);
for (j = 1; j < n; j++) 
{
if (n % j == 0) 
{
printf("%d", j);
if (j != n / 2) 
{
printf("+");
}}}
printf("=%d\n", n);
count++;
}}
return 0;
}
