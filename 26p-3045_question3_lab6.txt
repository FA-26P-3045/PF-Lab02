#include<stdio.h>
int main()
{
int n,i,j,k;
double x,s;
double factorial;
double power;


printf("Enter the x value: ");
scanf("%lf", &x);
printf("Enter the n value: ");
scanf("%d", &n);

s = 0;
for (k = 0; k <= n; k++) 
{
power = 1;
for (i = 0; i < k; i++) 
{
power = power * x;
}
factorial = 1;
for (j = 1; j <= k; j++) 
{
factorial = factorial * j;
}
s = s + (power / factorial);
}
printf("s = %.1f\n", s);
return 0;
}
