#include<stdio.h>
int main()
{
float weight;
float height;
float bmi;
 printf("Enter the weight in kg");
scanf("%f",&weight);
printf("Enter the height in meters");
scanf("%f",&height);

bmi = weight/(height*height);
printf("Computed BMI = %.2f\nCategory", bmi);

if(bmi<18.5){
printf("Underweight\n");
}
else if(bmi >= 18.5 && bmi<25.0)
{
printf("Normal\n");
}

else if(bmi>=25 && bmi<30.0)
{
printf("Overweight\n");}
else if (bmi>=30)
{
printf("Obese\n");
}
return 0;
}
