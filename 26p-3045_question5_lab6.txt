#include<stdio.h>
int main()
{
int rows,i,j;
char symbol;


printf("Enter the number of rows: ");
scanf("%d", &rows);
printf("Enter symbol: ");
scanf(" %c", &symbol);

for(i=1;i<=rows;i++)
{
for(j=1;j<=rows-i;j++)
{
printf(" ");
}
for(j=1;j<=(2*i-1);j++)
{
printf("%c", symbol);
}
printf("\n");
}
return 0;
}
